<script>

const SHEET_ID="1tAIAi6DCWFmNDnOoZlakb6ulI8S9yT1DX9qu7wjJAGQ";

const SHEET_NAME="MASTER";

const GVIZ="https://docs.google.com/spreadsheets/d/"+SHEET_ID+"/gviz/tq?tqx=out:json&sheet="+encodeURIComponent(SHEET_NAME);

function cell(c){ return c && c.v!=null ? c.v : ""; }

function stripPron(s){ return String(s).replace(/\s*\([^)]*\)/g,"").trim(); }

function parseYear(c){

  if(!c) return "";

  const raw=c.f!=null?c.f:c.v; if(raw==null) return "";

  const m=String(raw).match(/Date\((\d+),/); if(m) return m[1];

  const y=String(raw).match(/\b(19|20)\d{2}\b/); return y?y[0]:String(raw);

}

async function loadData(){

  const res=await fetch(GVIZ); const txt=await res.text();

  const json=JSON.parse(txt.replace(/^[^{]*\(/,"").replace(/\);?\s*$/,""));

  const rows=json.table.rows||[]; const entries=[];

  rows.forEach(r=>{

    const c=r.c||[];

    const nid=parseInt(cell(c[0]))||0;

    const comp=String(cell(c[1])).trim();

    const title=String(cell(c[2])).trim();

    if(!comp||!title||!(nid>0)) return;

    entries.push({ nid, c:comp, t:title,

      tracks:String(cell(c[3])).trim(), cd:String(cell(c[4])).trim(),

      label:String(cell(c[5])).trim(), spine:String(cell(c[6])).trim(),

      rid:String(cell(c[7])).trim(), s:stripPron(cell(c[8])),

      k:String(cell(c[9])).trim(), e:String(cell(c[10])).trim(),

      album:String(cell(c[11])).trim(), year:parseYear(c[12]) });

  });

  return entries;

}

function buildAlbums(entries){

  const by={}; entries.forEach(e=>{ (by[e.rid]=by[e.rid]||[]).push(e); });

  function ts(t){ const m=String(t||"").match(/\d+/); return m?parseInt(m[0]):999; }

  const albums=Object.keys(by).map(rid=>{

    const rows=by[rid].slice().sort((a,b)=>{ const cd=((parseInt(a.cd)||1)-(parseInt(b.cd)||1)); if(cd) return cd; return ts(a.tracks)-ts(b.tracks); });

    const a0=rows[0];

    return { rid, label:a0.label||"Unknown", spine:a0.spine||"", album:(rows.find(r=>r.album)||{}).album||"",

      tracks:rows.map(r=>({t:r.t,c:r.c,s:r.s,k:r.k,e:r.e,cd:r.cd})) };

  });

  albums.sort((A,B)=>{ const lc=(A.label||"").toLowerCase().localeCompare((B.label||"").toLowerCase()); if(lc) return lc; return ((parseInt(A.spine)||999999)-(parseInt(B.spine)||999999)); });

  return albums;

}

  function selectInBin(rid){ document.querySelectorAll('.bl-item').forEach(el=>el.classList.toggle('active',decodeURIComponent(el.dataset.rid)===rid));

    $('#binRight').innerHTML=renderAlbumCard(albumByRid[rid]); $('#binRight').scrollTop=0; }

  function openAlbumSolo(rid){ const a=albumByRid[rid]; if(!a) return; const bin=$('#bin'); bin.classList.add('solo');

    $('#binLeft').classList.add('hidden'); $('#binRight').innerHTML=renderAlbumCard(a); $('#binRight').scrollTop=0; openOverlay(); }

  function openOverlay(){ $('#overlay').classList.add('open'); document.body.style.overflow='hidden'; }

  function closeOverlay(){ $('#overlay').classList.remove('open'); document.body.style.overflow=''; }

  function showComposer(name){ mode='list'; syncToggle(); query=''; $('#q').value=''; composerSel=name; render(true); window.scrollTo({top:0,behavior:'smooth'}); }

  function clearComposer(){ composerSel=''; render(true); }

  function setRC(n,sing,plur){ const rc=$('#rc');

    if(composerSel&&mode==='list'){ rc.innerHTML='All works by <span class="chip-name">'+esc(composerSel)+'</span>'+

      '<button class="chip-clear" id="clearComp">clear ×</button> · '+n.toLocaleString()+' '+(n===1?sing:plur);

      const cc=$('#clearComp'); if(cc) cc.onclick=clearComposer; }

    else rc.textContent=n.toLocaleString()+' '+(n===1?sing:plur)+((query||composerSel)?' matching':''); }

  function render(reset){ mode==='list'?renderList(reset):renderShelf(); }

  function syncToggle(){ $('#tList').classList.toggle('active',mode==='list'); $('#tAlbums').classList.toggle('active',mode==='albums'); }

  $('#tList').onclick=()=>{ if(mode==='list')return; mode='list'; syncToggle(); render(true); window.scrollTo({top:0}); };

  $('#tAlbums').onclick=()=>{ if(mode==='albums')return; mode='albums'; composerSel=''; syncToggle(); render(true); window.scrollTo({top:0}); };

  let qt; $('#q').oninput=e=>{ clearTimeout(qt); qt=setTimeout(()=>{ query=e.target.value.trim().toLowerCase(); render(true); },150); };

  $('#chips').addEventListener('click',e=>{ const ch=e.target.closest('.chip'); if(!ch) return; const sc=ch.dataset.scope;

    if(scopes.has(sc)){ if(scopes.size>1){ scopes.delete(sc); ch.classList.remove('on'); } } else { scopes.add(sc); ch.classList.add('on'); }

    if(query) render(true); });

  $('#view').addEventListener('click',e=>{

    const h=e.target.closest('.grid-head .col'); if(h&&h.dataset.sort){ const k=h.dataset.sort;

      if(sortKey===k) sortDir*=-1; else { sortKey=k; sortDir=(k==='nid'?-1:1); } render(true); return; }

    const comp=e.target.dataset.comp; if(comp){ showComposer(comp); return; }

    const rid=e.target.dataset.rid; if(rid){ openAlbumSolo(rid); return; } });

  $('#binLeft').addEventListener('click',e=>{ const it=e.target.closest('.bl-item'); if(it) selectInBin(decodeURIComponent(it.dataset.rid)); });

  $('#binClose').addEventListener('click',closeOverlay);

  $('#overlay').addEventListener('click',e=>{ if(e.target.id==='overlay') closeOverlay(); });

  document.addEventListener('keydown',e=>{ if(e.key==='Escape') closeOverlay(); });

  new IntersectionObserver(es=>{ es.forEach(en=>{ if(en.isIntersecting&&mode==='list') renderList(false); }); },{rootMargin:'500px'}).observe($('#sentinel'));

  syncToggle(); render(true);

})();
